export function RichContent({ blocks }: { blocks: RichBlock[] }) {
  return (
    <div className="rich-content">
      {blocks.map((b, i) => {
        const Tag = (['h2', 'h3', 'p', 'ul', 'ol'].includes(b.tag) ? b.tag : 'p') as
          | 'h2'
          | 'h3'
          | 'p'
          | 'ul'
          | 'ol';
        const id = Tag === 'h2' ? headingId(b.html) : undefined;

        return <Tag id={id} key={i} dangerouslySetInnerHTML={{ __html: b.html }} />;
      })}
    </div>
  );
}

function plainText(html: string) {
  return html
    .replace(/<[^>]*>/g, '')
    .replace(/&amp;/g, '&')
    .trim();
}

function headingId(html: string) {
  return plainText(html)
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/(^-|-$)/g, '');
}

function formatDate(date: string) {
  return new Intl.DateTimeFormat('nl-NL', {
    day: 'numeric',
    month: 'long',
    year: 'numeric',
    timeZone: 'UTC',
  }).format(new Date(`${date}T12:00:00Z`));
}

function TableOfContents({ blocks }: { blocks: RichBlock[] }) {
  const headings = blocks.filter((block) => block.tag === 'h2');
  return (
    <details className="article-toc" open>
      <summary>Inhoudsopgave</summary>
      <ol>
        {headings.map((heading) => (
          <li key={heading.html}>
            <a href={`#${headingId(heading.html)}`}>{plainText(heading.html)}</a>
          </li>
        ))}
      </ol>
    </details>
  );
}

/** Airline pages the given product is registered for, e.g. to link a bag to its matching airlines. */
function ProductAirlineLinks({ project }: { project: Project }) {
  const matches = airlineRules.filter((rule) => project.airlineSlugs.includes(rule.slug));
  if (!matches.length) return null;

  return (
    <div className="rich-content">
      <h2>Past bij deze luchtvaartmaatschappijen</h2>
      <p>
        Deze buitenmaten zijn eerder vergeleken met de gepubliceerde cabineregels van:{' '}
        {matches.map((rule, index) => (
          <span key={rule.slug}>
            <Link href={`/luchtvaartmaatschappijen/${rule.slug}`}>Cabineregels {rule.name}</Link>
            {index < matches.length - 1 ? ', ' : '.'}
          </span>
        ))}{' '}
        Controleer bij je maatschappij altijd de actuele voorwaarden.
      </p>
    </div>
  );
}

/** Reistassen that are registered for this airline, shown as an extra, editorial section. */
function AirlineProductPicks({ slug }: { slug: string }) {
  const matches = projects.filter((project) => project.airlineSlugs.includes(slug)).slice(0, 3);
  if (!matches.length) return null;

  return (
    <section className="result-products" aria-labelledby="airline-products-title">
      <div>
        <span className="wizard-kicker">Reistassen</span>
        <h2 id="airline-products-title">Reistassen die bij deze buitenmaten passen</h2>
        <p>
          Deze reistassen zijn vergeleken met de hierboven genoemde maximale afmetingen. Doe de{' '}
          <Link href="/cabinecheck">cabinecheck</Link> voor een uitslag op basis van jouw huisdier
          en gewicht.
        </p>
      </div>
      <div className="result-product-grid">
        {matches.map((project) => (
          <ProjectCard key={project.id} project={project} placement="airline_page" />
        ))}
      </div>
    </section>
  );
}
