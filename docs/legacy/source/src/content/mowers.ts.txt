export function mowersForGuide(slug: string) {
  const selection: Record<string, string[]> = {
    'robotmaaier-zonder-draad': ['segway-navimow-i105e', 'worx-landroid-vision-m600', 'mova-1000'],
    'robotmaaier-kleine-tuin': [
      'husqvarna-aspire-r4',
      'gardena-sileno-minimo-500',
      'segway-navimow-i105e',
    ],
    'robotmaaier-grote-tuin': ['mammotion-luba-mini-2-awd-1000', 'eufy-e18'],
  };
  return (selection[slug] ?? []).flatMap((id) => mowers.filter((mower) => mower.slug === id));
}
