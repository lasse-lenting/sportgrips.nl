export type RichBlock = { tag: string; html: string };

  slug: string;
  title: string;
  description: string;
  location: string;
  date: string;
  modifiedAt?: string;
  updatedAt: string;

  hero: string;
  body: RichBlock[];
};

  slug: string;
  title: string;
  description: string;
  summary: string;

  image: string;

};
export type AirlineRule = {
  name: string;
  slug: string;
  cabinAllowed: boolean | null;
  dogAllowed: boolean | null;
  catAllowed: boolean | null;
  maxTotalWeight: number | null;
  maxLength: number | null;
  maxWidth: number | null;
  maxHeight: number | null;
  officialSource: string | null;
  checkedAt: string;
  verified: boolean;
  automaticRecommendations: boolean;
  manualReview: boolean;
  manualReviewReason: string | null;
  routeRestrictions: string[];
  breedRestrictions: string[];
};

const airlineEditorial = airlineEditorialData as Record<
  string,
  { verifiedAt: string; source: string; routeRestrictions: string[]; body: RichBlock[] }
>;
// Separate editorial additions survive the upstream travel-data sync.
export const events: CharityEvent[] = eventData.map((event) => {
  const editorial = airlineEditorial[event.slug];
  return editorial
    ? {
        ...event,
        modifiedAt: editorial.verifiedAt,
        updatedAt: new Intl.DateTimeFormat('nl-NL', {
          day: 'numeric',
          month: 'long',
          year: 'numeric',
          timeZone: 'UTC',
        }).format(new Date(`${editorial.verifiedAt}T12:00:00Z`)),
        body: editorial.body,
      }
    : event;
});
